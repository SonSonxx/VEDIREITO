local function stn() return Workspace:GetServerTimeNow() end

local calls = setmetatable({}, {__mode = "k"})
local function throttle(p, key, interval)
	local t = calls[p]
	if not t then t = {} calls[p] = t end
	local n = os.clock()
	if t[key] and n - t[key] < interval then return false end
	t[key] = n
	return true
end

local function aliveParts(p)
	local char = p.Character
	if not char then return nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if hum and hrp and hum.Health > 0 then return char, hum, hrp end
	return nil
end

local function defaultData()
	return {v = 1, coins = 0, level = 1, xp = 0, inventory = {"wood"}, equipped = "wood", kills = 0, dmg = 0,
		quests = {active = {}, done = {}}}
end

local function sanitize(raw)
	local d = defaultData()
	if type(raw) ~= "table" then return d end
	if type(raw.coins) == "number" then d.coins = math.clamp(math.floor(raw.coins), 0, 2e9) end
	if type(raw.level) == "number" then d.level = math.clamp(math.floor(raw.level), 1, Config.MaxLevel) end
	if type(raw.xp) == "number" then d.xp = math.max(0, math.floor(raw.xp)) end
	if type(raw.kills) == "number" then d.kills = math.max(0, math.floor(raw.kills)) end
	if type(raw.dmg) == "number" then d.dmg = math.max(0, math.floor(raw.dmg)) end
	if type(raw.inventory) == "table" then
		local seen = {wood = true}
		for _, id in ipairs(raw.inventory) do
			if type(id) == "string" and Config.SwordsById[id] and not seen[id] then
				seen[id] = true
				table.insert(d.inventory, id)
			end
		end
	end
	if type(raw.equipped) == "string" and table.find(d.inventory, raw.equipped) then d.equipped = raw.equipped end
	if type(raw.quests) == "table" then
		if type(raw.quests.active) == "table" then
			for id, prog in pairs(raw.quests.active) do
				local q = Config.QuestsById[id]
				if q and type(prog) == "number" then d.quests.active[id] = math.clamp(math.floor(prog), 0, q.count) end
			end
		end
		if type(raw.quests.done) == "table" then
			for id, v in pairs(raw.quests.done) do
				if Config.QuestsById[id] and v then d.quests.done[id] = true end
			end
		end
	end
	if d.level >= Config.MaxLevel then d.xp = 0 else d.xp = math.min(d.xp, Config.XpNeeded(d.level) - 1) end
	return d
end

local function snapshot(d)
	local act, done = {}, {}
	for k, v in pairs(d.quests.active) do act[k] = v end
	for k in pairs(d.quests.done) do done[k] = true end
	return {v = 1, coins = d.coins, level = d.level, xp = d.xp, inventory = table.clone(d.inventory), equipped = d.equipped,
		kills = d.kills, dmg = d.dmg, quests = {active = act, done = done}}
end

local function dataKey(p) return "Player_" .. p.UserId end

local function loadData(p)
	if not store then return false, nil end
	for i = 1, 3 do
		local ok, res = pcall(function() return store:GetAsync(dataKey(p)) end)
		if ok then return true, res end
		warn("[MigRPG] GetAsync falhou (" .. i .. "): " .. tostring(res))
		task.wait(1.5 * i)
	end
	return false, nil
end

local function saveData(p)
	local s = sessions[p]
	if not s or not s.canSave or not store then return false end
	local waited = 0
	while s.saving and waited < 10 do task.wait(0.2) waited += 0.2 end
	s.saving = true
	local snap = snapshot(s.data)
	local ok, err = false, nil
	for _ = 1, 3 do
		ok, err = pcall(function() store:UpdateAsync(dataKey(p), function() return snap end) end)
		if ok then break end
		task.wait(1.5)
	end
	s.saving = false
	if not ok then warn("[MigRPG] Falha ao salvar " .. p.Name .. ": " .. tostring(err)) end
	return ok
end

-- ============================== ESTADO ==============================
pushState = function(p)
	local s = sessions[p]
	if not s or not s.ready then return end
	local d = s.data
	local act, done = {}, {}
	for k, v in pairs(d.quests.active) do act[k] = v end
	for k in pairs(d.quests.done) do done[k] = true end
	rState:FireClient(p, {
		coins = d.coins, level = d.level, xp = d.xp,
		need = (d.level >= Config.MaxLevel) and 0 or Config.XpNeeded(d.level),
		inv = table.clone(d.inventory), equipped = d.equipped, kills = d.kills, dmg = d.dmg,
		active = act, done = done, admin = isAdmin(p), skillReady = s.skillCd[d.equipped] or 0,
		canSave = s.canSave, speedMult = s.speedMult,
	})
end

local function applyMaxHealth(p, heal)
	local s = sessions[p]
	local char = p.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not (s and hum) then return end
	local mh = Config.MaxHealth(s.data.level)
	local frac = hum.Health / math.max(1, hum.MaxHealth)
	hum.MaxHealth = mh
	hum.Health = heal and mh or math.clamp(frac * mh, 1, mh)
end

local function addXp(p, amount)
	local s = sessions[p]
	if not s then return end
	local d = s.data
	if d.level >= Config.MaxLevel then return end
	d.xp += amount
	local leveled = false
	while d.level < Config.MaxLevel and d.xp >= Config.XpNeeded(d.level) do
		d.xp -= Config.XpNeeded(d.level)
		d.level += 1
		leveled = true
	end
	if d.level >= Config.MaxLevel then d.xp = 0 end
	if leveled then
		applyMaxHealth(p, true)
		notify(p, "NÍVEL " .. d.level .. "!  Vida máxima aumentada.", "level")
		local _, _, hrp = aliveParts(p)
		if hrp then rFX:FireAllClients({k = "levelup", p = hrp.Position}) end
	end
end

local function setLevel(p, level)
	local s = sessions[p]
	if not s then return end
	s.data.level = math.clamp(math.floor(level), 1, Config.MaxLevel)
	s.data.xp = 0
	applyMaxHealth(p, true)
	pushState(p)
end

-- ============================== QUESTS ==============================
local function questMatches(q, kind)
	if not q.kinds then return true end
	return table.find(q.kinds, kind) ~= nil
end

local function questOnKill(p, kind)
	local s = sessions[p]
	if not s then return end
	for id, prog in pairs(s.data.quests.active) do
		local q = Config.QuestsById[id]
		if q and prog < q.count and questMatches(q, kind) then
			s.data.quests.active[id] = prog + 1
			if prog + 1 >= q.count then
				notify(p, "Missão pronta para entregar: " .. q.name, "quest")
			else
				notify(p, ("%s: %d/%d"):format(q.name, prog + 1, q.count), "quest")
			end
		end
	end
end

local function countActive(d)
	local n = 0
	for _ in pairs(d.quests.active) do n += 1 end
	return n
end

local function acceptQuest(p, qid)
	local d = sessions[p].data
	local q = Config.QuestsById[qid]
	if not q or d.quests.active[qid] ~= nil then return false, "Missão já ativa." end
	if d.quests.done[qid] and not q.repeatable then return false, "Missão já concluída." end
	if countActive(d) >= Config.MaxActiveQuests then return false, "Você já tem missões demais ativas." end
	d.quests.active[qid] = 0
	return true, "Missão aceita: " .. q.name .. ". " .. q.desc
end

local function completeQuest(p, qid)
	local d = sessions[p].data
	local q = Config.QuestsById[qid]
	if not q then return false, "Missão inválida." end
	local prog = d.quests.active[qid]
	if prog == nil or prog < q.count then return false, "Você ainda não terminou esta missão." end
	d.quests.active[qid] = nil
	if not q.repeatable then d.quests.done[qid] = true end
	d.coins += q.coins
	addXp(p, q.xp)
	notify(p, ("Missão concluída! +%d moedas extras  +%d XP"):format(q.coins, q.xp), "gain")
	return true, ("Excelente trabalho! Aqui está sua recompensa: %d moedas e %d XP."):format(q.coins, q.xp)
end

local function buildTool(def)
	local tool = Instance.new("Tool")
	tool.Name = def.name
	tool.RequiresHandle = true
	tool.CanBeDropped = false
	tool.ToolTip = def.name
	tool:SetAttribute("MigSword", def.id)
	local handle = Instance.new("Part")
	handle.Name = "Handle"
	handle.Size = Vector3.new(0.6, 0.6, 1.8)
	handle.Color = def.hilt
	handle.Material = Enum.Material.Wood
	handle.CanCollide = false
	handle.Parent = tool
	local function add(size, offset, color, mat, shape)
		local p = Instance.new("Part")
		p.Size = size
		p.Color = color
		p.Material = mat
		p.CanCollide = false
		p.Massless = true
		p.CanQuery = false
		if shape then p.Shape = shape end
		local w = Instance.new("Weld")
		w.Part0 = handle
		w.Part1 = p
		w.C0 = offset
		w.C1 = CFrame.new()
		w.Parent = p
		p.Parent = tool
		return p
	end
	local len = def.len
	add(Vector3.new(0.22, 0.85, len), CFrame.new(0, 0, -(0.9 + len / 2)), def.blade, Enum.Material.Metal)
	add(Vector3.new(2, 0.35, 0.5), CFrame.new(0, 0, -0.95), Color3.fromRGB(212, 172, 60), Enum.Material.Metal)
	add(Vector3.new(0.7, 0.7, 0.7), CFrame.new(0, 0, 0.95), Color3.fromRGB(212, 172, 60), Enum.Material.Metal, Enum.PartType.Ball)
	if def.glow then
		local edge = add(Vector3.new(0.27, 0.5, len - 0.3), CFrame.new(0, 0, -(0.9 + len / 2)), def.glow, Enum.Material.Neon)
		edge.Transparency = 0.35
		local rar = Config.Rarities[def.rarity]
		if rar and rar.order >= 3 then
			local l = Instance.new("PointLight")
			l.Color = def.glow
			l.Brightness = 1.2
			l.Range = 10
			l.Parent = edge
		end
		if rar and rar.order >= 4 then
			local e = Instance.new("ParticleEmitter")
			e.Color = ColorSequence.new(def.glow)
			e.Rate = 8
			e.Lifetime = NumberRange.new(0.6, 1)
			e.Speed = NumberRange.new(0.5, 1.5)
			e.Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.3), NumberSequenceKeypoint.new(1, 0)})
			e.LightEmission = 1
			e.Parent = edge
		end
	end
	return tool
end

local function refreshTool(p)
	local s = sessions[p]
	if not s then return end
	local def = Config.SwordsById[s.data.equipped]
	if not def then return end
	local char = p.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local backpack = p:FindFirstChildOfClass("Backpack") or p:WaitForChild("Backpack", 5)
	if not backpack then return end
	for _, container in ipairs({backpack, char}) do
		if container then
			for _, c in ipairs(container:GetChildren()) do
				if c:IsA("Tool") and c:GetAttribute("MigSword") then c:Destroy() end
			end
		end
	end
	local tool = buildTool(def)
	tool.Parent = backpack
	if hum and hum.Health > 0 then hum:EquipTool(tool) end
end

local function recsInRadius(pos, radius)
	local out = {}
	for _, rec in ipairs(Slimes) do
		if not rec.dead and rec.body.Parent then
			local d = (rec.body.Position - pos).Magnitude - rec.radius
			if d <= radius then
				rec.tmpD = d
				table.insert(out, rec)
			end
		end
	end
	table.sort(out, function(a, b) return a.tmpD < b.tmpD end)
	return out
end

local function setSleep(rec, sleep)
	if rec.dead or rec.sleep == sleep then return end
	rec.sleep = sleep
	rec.body.Anchored = sleep
	if not sleep then pcall(function() rec.body:SetNetworkOwner(nil) end) end
end

local function giveKillRewards(rec, killer)
	local cfg = rec.cfg
	local xpBase = (8 + rec.level * 6) * cfg.xpMul
	local coinBase = (4 + rec.level * 3) * cfg.coinMul
	for p, dmg in pairs(rec.dmgBy) do
		local s = sessions[p]
		if p.Parent == Players and s and s.ready and (dmg >= rec.maxHp * 0.12 or p == killer) then
			local d = s.data
			local f = math.clamp(1 - math.max(0, d.level - rec.level - 5) * 0.1, 0.2, 1)
			local xp = math.floor(xpBase * f + 0.5)
			local coins = math.floor(coinBase * f + 0.5)
			d.coins += coins
			d.kills += 1
			addXp(p, xp)
			notify(p, ("+%d XP   +%d moedas"):format(xp, coins), "gain")
			questOnKill(p, rec.kind)
			if rec.kind == "ReiSlime" then
				if not table.find(d.inventory, "king") then
					table.insert(d.inventory, "king")
					notify(p, "DROP MÍTICO: Lâmina do Rei Slime!", "level")
				else
					d.coins += 2500
					notify(p, "Rei Slime derrotado! +2.500 moedas de bônus.", "gain")
				end
			end
			pushState(p)
		end
	end
end

local function killSlime(rec, killer)
	if rec.dead then return end
	rec.dead = true
	rec.hp = 0
	local idx = table.find(Slimes, rec)
	if idx then table.remove(Slimes, idx) end
	if rec.body.Parent then
		rFX:FireAllClients({k = "burst", p = rec.body.Position, c = rec.color, n = 18})
	end
	giveKillRewards(rec, killer)
	rec.model:Destroy()
	local sp = rec.spawn
	if sp then
		sp.alive = nil
		sp.respawnAt = os.clock() + sp.respawn
	end
end

local function damageSlime(rec, amount, p, crit)
	if rec.dead then return 0 end
	amount = math.min(amount, rec.hp)
	rec.hp -= amount
	if p then
		local s = sessions[p]
		rec.dmgBy[p] = (rec.dmgBy[p] or 0) + amount
		if s then s.data.dmg += amount end
		rFX:FireClient(p, {k = "dmg", p = rec.body.Position + Vector3.new(0, rec.radius + 1, 0), a = math.floor(amount), c = crit})
		rec.forcedTarget = p
		rec.forcedUntil = os.clock() + 6
	end
	updateSlimeGui(rec)
	if rec.hp <= 0 then killSlime(rec, p) end
	return amount
end

local function knock(rec, fromPos, strength)
	if rec.dead then return end
	setSleep(rec, false)
	local dir = (rec.body.Position - fromPos) * Vector3.new(1, 0, 1)
	dir = dir.Magnitude > 0.1 and dir.Unit or Vector3.new(0, 0, 1)
	local m = rec.body.AssemblyMass
	rec.body:ApplyImpulse(Vector3.new(dir.X * strength * m, strength * 0.6 * m, dir.Z * strength * m))
	rec.lastHop = os.clock()
end

local function spawnSlime(sp)
	local cfg = Config.SlimeKinds[sp.kind] or Config.SlimeKinds.Slime
	local level = math.clamp(sp.level or 1, 1, 100)
	local size = cfg.size
	local color = sp.color or cfg.color
	local title = sp.title or cfg.name
	local maxHp = math.floor((30 + 20 * level ^ 1.2) * cfg.hpMul)

	local model = Instance.new("Model")
	model.Name = title
	local body = Instance.new("Part")
	body.Name = "Body"
	body.Shape = Enum.PartType.Ball
	body.Size = Vector3.new(size, size, size)
	body.Color = color
	body.Material = Enum.Material.SmoothPlastic
	body.Transparency = 0.2
	body.CustomPhysicalProperties = PhysicalProperties.new(0.7, 1.2, 0, 1, 1)
	body.TopSurface = Enum.SurfaceType.Smooth
	body.BottomSurface = Enum.SurfaceType.Smooth
	body.CFrame = CFrame.new(sp.pos + Vector3.new(0, size / 2 + 1.5, 0))
	body.Parent = model

	local function attach(part, offset)
		part.CanCollide = false
		part.CanQuery = false
		part.Massless = true
		part.Anchored = false
		local w = Instance.new("Weld")
		w.Part0 = body
		w.Part1 = part
		w.C0 = offset
		w.C1 = CFrame.new()
		w.Parent = part
		part.Parent = model
	end
	local core = Instance.new("Part")
	core.Shape = Enum.PartType.Ball
	core.Size = Vector3.new(size * 0.5, size * 0.5, size * 0.5)
	core.Color = Color3.new(color.R * 0.55, color.G * 0.55, color.B * 0.55)
	core.Material = Enum.Material.SmoothPlastic
	attach(core, CFrame.new(0, -size * 0.1, 0))
	for _, sx in ipairs({-1, 1}) do
		local eye = Instance.new("Part")
		eye.Shape = Enum.PartType.Ball
		eye.Size = Vector3.new(size * 0.16, size * 0.16, size * 0.16)
		eye.Color = Color3.fromRGB(20, 20, 30)
		eye.Material = Enum.Material.SmoothPlastic
		attach(eye, CFrame.new(sx * size * 0.2, size * 0.12, -size * 0.43))
	end

	local att = Instance.new("Attachment")
	att.Parent = body
	local ao = Instance.new("AlignOrientation")
	ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
	ao.Attachment0 = att
	ao.RigidityEnabled = true
	ao.CFrame = CFrame.new()
	ao.Parent = body

	local rec = {
		model = model, body = body, ao = ao, cfg = cfg, kind = sp.kind, level = level, title = title, color = color,
		hp = maxHp, maxHp = maxHp, radius = size / 2, home = sp.pos, spawn = sp.temp and nil or sp,
		dmg = math.floor((3 + level * 1.4) * cfg.dmgMul), dmgBy = {}, dead = false, sleep = false,
		lastHop = 0, nextHop = os.clock() + rng:NextNumber(0, 1), atkReady = 0, slowUntil = 0, slowAmt = 0,
		forcedUntil = 0, nextWander = 0,
	}
	buildSlimeGui(rec)
	model:SetAttribute("Slime", true)
	model:SetAttribute("Kind", sp.kind)
	model:SetAttribute("Level", level)
	model.PrimaryPart = body
	model.Parent = slimeFolder
	pcall(function() body:SetNetworkOwner(nil) end)
	table.insert(Slimes, rec)
	if sp.alive ~= nil or not sp.temp then sp.alive = rec end
	return rec
end

local function nearestPlayer(pos, maxDist)
	local best, bd = nil, maxDist
	for _, p in ipairs(Players:GetPlayers()) do
		local _, _, hrp = aliveParts(p)
		if hrp then
			local d = (hrp.Position - pos).Magnitude
			if d < bd then best, bd = p, d end
		end
	end
	return best, bd
end

local function poisonPlayer(target, perTick)
	task.spawn(function()
		for _ = 1, 3 do
			task.wait(1)
			local _, hum = aliveParts(target)
			if not hum then return end
			hum:TakeDamage(perTick)
		end
	end)
end

local function grounded(rec)
	return Workspace:Raycast(rec.body.Position, Vector3.new(0, -(rec.radius + 0.7), 0), rayParams) ~= nil
end

local function hop(rec, dir, mult)
	local cfg = rec.cfg
	local t = os.clock()
	local sf = (t < rec.slowUntil) and (1 - rec.slowAmt) or 1
	local m = rec.body.AssemblyMass
	local sp = cfg.speed * sf * (mult or 1)
	rec.ao.CFrame = CFrame.lookAt(Vector3.zero, dir)
	rec.body:ApplyImpulse(Vector3.new(dir.X * sp * m, cfg.jump * m, dir.Z * sp * m))
	rec.lastHop = t
end

local function updateSlime(rec, dt, t)
	local body = rec.body
	if not body.Parent then killSlime(rec, nil) return end
	local pos = body.Position
	local cfg = rec.cfg

	local near, nd = nearestPlayer(pos, 280)
	if not near then setSleep(rec, true) return end
	setSleep(rec, false)

	if rec.dot then
		if t >= rec.dot.untilT then
			rec.dot = nil
		elseif t >= rec.dot.nextTick then
			rec.dot.nextTick = t + 0.5
			damageSlime(rec, math.max(1, math.floor(rec.dot.dps * 0.5)), rec.dot.src, false)
			if rec.dead then return end
		end
	end

	if pos.Y < -120 or (pos - rec.home).Magnitude > 450 then
		body.AssemblyLinearVelocity = Vector3.zero
		body.CFrame = CFrame.new(rec.home + Vector3.new(0, 6, 0))
		return
	end

	local homeDist = (pos - rec.home).Magnitude
	local target = rec.target
	if target then
		local _, _, thrp = aliveParts(target)
		if not thrp or target.Parent ~= Players or (thrp.Position - pos).Magnitude > cfg.aggro * 1.7 or homeDist > cfg.leash + 25 then
			target = nil
			rec.target = nil
		end
	end
	if not target and homeDist < cfg.leash then
		if rec.forcedTarget and t < rec.forcedUntil and aliveParts(rec.forcedTarget) then
			target = rec.forcedTarget
		else
			local p = nearestPlayer(pos, cfg.aggro)
			target = p
		end
		rec.target = target
	end

	local isGrounded = grounded(rec)
	if isGrounded and t - rec.lastHop > 0.3 then
		local v = body.AssemblyLinearVelocity
		body.AssemblyLinearVelocity = Vector3.new(0, math.min(v.Y, 0), 0)
	end

	if target then
		local _, thum, thrp = aliveParts(target)
		if thrp then
			local to = thrp.Position - pos
			local flat = Vector3.new(to.X, 0, to.Z)
			local dist = flat.Magnitude
			local dir = dist > 0.1 and flat.Unit or Vector3.new(0, 0, -1)
			rec.ao.CFrame = CFrame.lookAt(Vector3.zero, dir)
			if dist <= rec.radius + 3.4 and math.abs(to.Y) < rec.radius + 6 then
				if t >= rec.atkReady then
					rec.atkReady = t + cfg.atkCd
					thum:TakeDamage(rec.dmg)
					rFX:FireClient(target, {k = "hurt"})
					local m = body.AssemblyMass
					body:ApplyImpulse(Vector3.new(dir.X * 12 * m, 18 * m, dir.Z * 12 * m))
					rec.lastHop = t
					if cfg.poison then poisonPlayer(target, math.max(1, math.floor(rec.dmg * 0.25))) end
				end
			elseif isGrounded and t >= rec.nextHop then
				local sf = (t < rec.slowUntil) and (1 - rec.slowAmt) or 1
				hop(rec, dir, 1)
				rec.nextHop = t + 0.75 / sf
			end
		end
	else
		rec.hp = math.min(rec.maxHp, rec.hp + rec.maxHp * 0.04 * dt)
		updateSlimeGui(rec)
		if isGrounded and t >= rec.nextHop then
			local dir
			if homeDist > 22 then
				local to = rec.home - pos
				dir = Vector3.new(to.X, 0, to.Z).Unit
			elseif rng:NextNumber() < 0.55 then
				local a = rng:NextNumber(0, math.pi * 2)
				dir = Vector3.new(math.cos(a), 0, math.sin(a))
			end
			if dir then hop(rec, dir, 0.55) end
			rec.nextHop = t + rng:NextNumber(1.2, 2.6)
		end
	end
end

task.spawn(function()
	local last = os.clock()
	while true do
		task.wait(0.12)
		local t = os.clock()
		local dt = t - last
		last = t
		for _, rec in ipairs(table.clone(Slimes)) do
			if not rec.dead then
				local ok, err = pcall(updateSlime, rec, dt, t)
				if not ok then warn("[MigRPG] erro IA slime: " .. tostring(err)) end
			end

	end
end)

-- coleta de pontos de spawn do mapa (markers da pasta Inimigos)
local function collectSpawns()
	local inim = MIG:WaitForChild("Inimigos", 10)
	if not inim then return end
	local caveFloorY
	local cav = MIG:FindFirstChild("Cavernas")
	local interior = cav and cav:FindFirstChild("Interior_Cristalino")
	local rune = interior and interior:FindFirstChild("CirculoRunico")
	if rune then caveFloorY = rune:GetPivot().Position.Y end
	for _, d in ipairs(inim:GetDescendants()) do
		if d:IsA("BasePart") and d:GetAttribute("Tipo") then
			local tipo = d:GetAttribute("Tipo")
			local kind = tipo
			if tipo == "Chefe" then kind = (d.Name == "BossSpawn_ReiSlime") and "ReiSlime" or "Chefe" end
			if not Config.SlimeKinds[kind] then kind = "Slime" end
			local pos = d.Position
			local parentName = d.Parent and d.Parent.Name or ""
			if caveFloorY and (parentName:find("^SlimeCristal_") or parentName:find("^ChefeCaverna")) then
				local r = Workspace:Raycast(Vector3.new(pos.X, caveFloorY + 12, pos.Z), Vector3.new(0, -60, 0), terrainParams)
				pos = r and r.Position or Vector3.new(pos.X, caveFloorY + 1, pos.Z)
			end
			local color = d:GetAttribute("Cor")
			table.insert(Spawns, {
				kind = kind, level = d:GetAttribute("Nivel") or 1, pos = pos, respawn = d:GetAttribute("Respawn") or 20,
				color = typeof(color) == "Color3" and color or nil,
				title = Config.NamedBosses[d.Name] or Config.NamedBosses[parentName], respawnAt = 0,
			})
		end
	end
end

-- ============================== COMBATE =============================
local function flatLook(hrp)
	local v = hrp.CFrame.LookVector
	v = Vector3.new(v.X, 0, v.Z)
	if v.Magnitude < 0.01 then return Vector3.new(0, 0, -1) end
	return v.Unit
end

local function calcDamage(p, mult)
	local s = sessions[p]
	local def = Config.SwordsById[s.data.equipped]
	local base = def.damage * Config.DamageScale(s.data.level) * (mult or 1) * rng:NextNumber(0.9, 1.1)
	local crit = rng:NextNumber() < def.crit
	if crit then base *= 1.8 end
	return math.max(1, math.floor(base + 0.5)), crit
end

local function scaledBase(p)
	local s = sessions[p]
	return Config.SwordsById[s.data.equipped].damage * Config.DamageScale(s.data.level)
end

local function applyDot(rec, p, spec)
	if not spec then return end
	rec.dot = {untilT = os.clock() + spec.dur, dps = scaledBase(p) * spec.mult, src = p, nextTick = os.clock() + 0.5}
end

local function playToolAnim(char, name)
	local tool = char:FindFirstChildWhichIsA("Tool")
	if tool and tool:GetAttribute("MigSword") then
		local old = tool:FindFirstChild("toolanim")
		if old then old:Destroy() end
		local sv = Instance.new("StringValue")
		sv.Name = "toolanim"
		sv.Value = name
		sv.Parent = tool
	end
end

local function performAttack(p)
	local s = sessions[p]
	if not (s and s.ready) then return end
	local char, hum, hrp = aliveParts(p)
	if not hrp then return end
	local def = Config.SwordsById[s.data.equipped]
	local t = os.clock()
	if t < s.attackReady then return end
	s.attackReady = t + def.cooldown
	playToolAnim(char, "Slash")
	local look = flatLook(hrp)
	rFX:FireAllClients({k = "slash", cf = hrp.CFrame * CFrame.new(0, 0.5, -3), c = def.glow or def.blade, s = def.range / 9})
	local hits = 0
	for _, rec in ipairs(recsInRadius(hrp.Position, def.range)) do
		local to = rec.body.Position - hrp.Position
		local flat = Vector3.new(to.X, 0, to.Z)
		if flat.Magnitude < rec.radius + 1 or flat.Unit:Dot(look) > 0.25 then
			if math.abs(to.Y) < 12 + rec.radius then
				local dmg, crit = calcDamage(p, 1)
				damageSlime(rec, dmg, p, crit)
				knock(rec, hrp.Position, 14)
				hits += 1
				if hits >= 3 then break end
			end
		end
	end
end

Skills.dash = function(p, s, def, sk, char, hum, hrp)
	local dir = flatLook(hrp)
	local start = hrp.Position
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = {char, slimeFolder}
	local res = Workspace:Raycast(start, dir * sk.dist, rp)
	local dist = res and math.max(0, res.Distance - 2.5) or sk.dist
	local goal = start + dir * dist
	rFX:FireAllClients({k = "dash", a = start, b = goal, c = sk.color})
	hrp.CFrame = CFrame.lookAt(goal, goal + dir)
	for _, rec in ipairs(recsInRadius(start + dir * (dist / 2), dist / 2 + sk.width)) do
		local to = rec.body.Position - start
		local along = to:Dot(dir)
		local lateral = (to - dir * along) * Vector3.new(1, 0, 1)
		if along > -3 and along < dist + 3 and lateral.Magnitude <= sk.width / 2 + rec.radius then
			local dmg, crit = calcDamage(p, sk.mult)
			damageSlime(rec, dmg, p, crit)
			knock(rec, start, 20)
		end
	end
	return true
end

	for _, rec in ipairs(recsInRadius(start, sk.length + 6)) do

			local dmg, crit = calcDamage(p, sk.mult)
			damageSlime(rec, dmg, p, crit)
			applyDot(rec, p, sk.dot)

	return true
end

Skills.chain = function(p, s, def, sk, char, hum, hrp)
	local first = recsInRadius(hrp.Position, sk.range)[1]
	if not first then return false, "Nenhum alvo por perto." end
	local hit = {}
	local cur = first
	local prev = hrp.Position + Vector3.new(0, 2, 0)
	for _ = 1, sk.targets do
		if not cur then break end
		hit[cur] = true
		local cpos = cur.body.Position
		rFX:FireAllClients({k = "bolt", a = prev, b = cpos, c = sk.color})
		local dmg, crit = calcDamage(p, sk.mult)
		damageSlime(cur, dmg, p, crit)
		prev = cpos
		local nxt = nil
		for _, r2 in ipairs(recsInRadius(cpos, 18)) do
			if not hit[r2] then nxt = r2 break end
		end
		cur = nxt
	end
	return true
end

local function castSkill(p)
	local s = sessions[p]
	if not (s and s.ready) then return end
	local char, hum, hrp = aliveParts(p)
	if not hrp then return end
	local def = Config.SwordsById[s.data.equipped]
	local sk = def.skill
	if stn() < (s.skillCd[def.id] or 0) then return end
	local fn = Skills[sk.kind]
	if not fn then return end
	local ok, used, msg = pcall(fn, p, s, def, sk, char, hum, hrp)
	if not ok then warn("[MigRPG] erro na habilidade: " .. tostring(used)) return end
	if used == false then
		if msg then notify(p, msg, "warn") end
		return
	end
	playToolAnim(char, "Lunge")
	s.skillCd[def.id] = stn() + sk.cd
	pushState(p)
end

local function buildDialogue(p, npcName, text)
	local cfg = Config.NPCs[npcName] or {text = "Olá, viajante!"}
	local d = sessions[p].data
	local options = {}
	if cfg.quests then
		for _, qid in ipairs(cfg.quests) do
			local q = Config.QuestsById[qid]
			local prog = d.quests.active[qid]
			if prog ~= nil then
				if prog >= q.count then
					table.insert(options, {id = "turn:" .. qid, label = "Entregar missão: " .. q.name})
				else
					table.insert(options, {id = "info:" .. qid, label = ("Em andamento: %s (%d/%d)"):format(q.name, prog, q.count)})
				end
			elseif q.repeatable or not d.quests.done[qid] then
				table.insert(options, {id = "accept:" .. qid, label = "Aceitar missão: " .. q.name})
			end
		end
	end
	if cfg.shop then table.insert(options, {id = "shop", label = "Ver espadas", close = true}) end
	if cfg.rest then table.insert(options, {id = "rest", label = ("Descansar (%d moedas) — curar vida"):format(Config.RestCost)}) end
	if cfg.tip then table.insert(options, {id = "tip", label = "Preciso de uma dica"}) end
	table.insert(options, {id = "bye", label = "Até logo", close = true})
	return {npc = npcName, text = text or cfg.text, options = options, pos = npcPos[npcName]}
end

local function openDialogue(p, npcName)
	local s = sessions[p]
	if not (s and s.ready) then return end
	local pos = npcPos[npcName]
	local _, _, hrp = aliveParts(p)
	if not pos or not hrp or (hrp.Position - pos).Magnitude > 30 then return end
	rDialogue:FireClient(p, buildDialogue(p, npcName))
end

rChoice.OnServerEvent:Connect(function(p, npcName, optId)
	if typeof(npcName) ~= "string" or typeof(optId) ~= "string" then return end
	if not throttle(p, "dlg", 0.2) then return end
	local s = sessions[p]
	if not (s and s.ready) then return end
	local pos = npcPos[npcName]
	local char, hum, hrp = aliveParts(p)
	if not pos or not hrp or (hrp.Position - pos).Magnitude > 30 then return end
	local valid = false
	for _, o in ipairs(buildDialogue(p, npcName).options) do
		if o.id == optId then valid = true break end
	end
	if not valid then return end
	local cfg = Config.NPCs[npcName] or {}
	local kind, arg = optId:match("^(%w+):?(.*)$")
	local reply = nil
	if kind == "accept" then
		local ok, msg = acceptQuest(p, arg)
		reply = msg
	elseif kind == "turn" then
		local ok, msg = completeQuest(p, arg)
		reply = msg
	elseif kind == "info" then
		local q = Config.QuestsById[arg]
		reply = q and (q.desc .. " Volte quando terminar.") or "..."
	elseif kind == "shop" then
		rOpenShop:FireClient(p)
		return
	elseif kind == "rest" then
		if s.data.coins < Config.RestCost then
			reply = "Você não tem moedas suficientes. Volte quando tiver."
		else
			s.data.coins -= Config.RestCost
			hum.Health = hum.MaxHealth
			reply = "Pronto! Você está novinho em folha."
		end
	elseif kind == "tip" then
		reply = cfg.tip or "Boa sorte na jornada!"
	elseif kind == "bye" then
		return
	end
	pushState(p)
	rDialogue:FireClient(p, buildDialogue(p, npcName, reply))
end)

RunService.Heartbeat:Connect(function(dt)
	local t = os.clock()
	for p, s in pairs(sessions) do
		if s.ready then
			local _, hum, hrp = aliveParts(p)
			if hum then
				local moving = hum.MoveDirection.Magnitude > 0.1
				local exhausted = t < s.exhaustedUntil
				local canSprint = (not exhausted) and (not s.needRecover) and s.stamina > 0
				local sprinting = s.wantSprint and canSprint and moving
				if sprinting then
					s.stamina = math.max(0, s.stamina - Config.SPRINT_DRAIN * dt)
					s.regenAt = t + Config.SPRINT_STOP_DELAY
					if s.stamina <= 0 then
						s.exhaustedUntil = t + Config.SPRINT_EXHAUST_DELAY
						s.regenAt = s.exhaustedUntil
						s.needRecover = true
						s.wantSprint = false
					end
				elseif t >= s.regenAt then
					s.stamina = math.min(100, s.stamina + Config.SPRINT_REGEN * dt)
					if s.needRecover and s.stamina >= 15 then s.needRecover = false end
				end
				local target = Config.WALK_SPEED * s.speedMult * (sprinting and (Config.SPRINT_SPEED / Config.WALK_SPEED) or 1)
				if math.abs(hum.WalkSpeed - target) > 0.01 then hum.WalkSpeed = target end
				local flag = sprinting and 1 or ((t < s.exhaustedUntil) and 2 or 0)
				local remain = math.max(0, s.exhaustedUntil - t)
				if t - s.lastSend > 0.1 and (math.abs(s.stamina - s.lastSentVal) >= 0.5 or flag ~= s.lastSentFlag or flag == 2) then
					s.lastSend = t
					s.lastSentVal = s.stamina
					s.lastSentFlag = flag
					rStamina:FireClient(p, s.stamina, flag, remain)
				end
			end
		end
	end
end)

local function findPlayer(name)
	name = string.lower(name)
	for _, p in ipairs(Players:GetPlayers()) do
		if string.lower(p.Name) == name then return p end
	end
	for _, p in ipairs(Players:GetPlayers()) do
		if string.lower(p.Name):sub(1, #name) == name or string.lower(p.DisplayName):sub(1, #name) == name then return p end
	end
	return nil
end

	if name == nil or name == "" then return {admin} end
	if string.lower(name) == "all" or string.lower(name) == "todos" then return Players:GetPlayers() end
	local p = findPlayer(name)
	return p and {p} or {}
end

	local num = typeof(args.value) == "number" and args.value or nil

	if cmd == "heal" then
		for _, p in ipairs(targets(admin, tname)) do
			local _, hum = aliveParts(p)
			if hum then hum.Health = hum.MaxHealth end
		end
		notify(admin, "Cura aplicada.", "ok")
	elseif cmd == "speed" then
		if not num then return end
		num = math.clamp(num, 0.5, 5)
		for _, p in ipairs(targets(admin, tname)) do
			local s = sessions[p]
			if s then s.speedMult = num pushState(p) end
		end
		notify(admin, "Velocidade ×" .. num, "ok")
	elseif cmd == "coins" then
		if not num or (args.mode ~= "add" and args.mode ~= "set") then return end
		num = math.clamp(math.floor(num), -1e9, 1e9)
		for _, p in ipairs(targets(admin, tname)) do
			local s = sessions[p]
			if s then
				s.data.coins = math.clamp((args.mode == "set") and num or (s.data.coins + num), 0, 2e9)
				pushState(p)
			end
		end
		notify(admin, "Moedas atualizadas.", "ok")
	elseif cmd == "level" then
		if not num then return end
		for _, p in ipairs(targets(admin, tname)) do setLevel(p, num) end
		notify(admin, "Nível definido.", "ok")
	elseif cmd == "give_sword" or cmd == "remove_sword" then
		local id = args.sword
		if typeof(id) ~= "string" or not Config.SwordsById[id] then return end
		for _, p in ipairs(targets(admin, tname)) do
			local s = sessions[p]
			if s then
				local idx = table.find(s.data.inventory, id)
				if cmd == "give_sword" and not idx then
					table.insert(s.data.inventory, id)
				elseif cmd == "remove_sword" and idx and id ~= "wood" then
					table.remove(s.data.inventory, idx)
					if s.data.equipped == id then s.data.equipped = "wood" refreshTool(p) end
				end
				pushState(p)
			end
		end
		notify(admin, "Inventário atualizado.", "ok")
	elseif cmd == "tp_poi" then
		local name = args.name
		local folder = MIG:FindFirstChild("PontosDeInteresse")
		local marker = typeof(name) == "string" and folder and folder:FindFirstChild("POI_" .. name)
		local _, _, hrp = aliveParts(admin)
		if marker and hrp then
			hrp.CFrame = CFrame.new(marker.Position + Vector3.new(0, 6, 0))
			notify(admin, "Teleportado: " .. name, "ok")
		end
	elseif cmd == "tp_to" or cmd == "bring" then
		local other = findPlayer(tname)
		if not other or other == admin then notify(admin, "Jogador não encontrado.", "err") return end
		local _, _, ah = aliveParts(admin)
		local _, _, oh = aliveParts(other)
		if ah and oh then
			if cmd == "tp_to" then ah.CFrame = oh.CFrame * CFrame.new(0, 0, 5) else oh.CFrame = ah.CFrame * CFrame.new(0, 0, -5) end
		end
	elseif cmd == "slimes_kill" then
		local n = 0
		for _, rec in ipairs(table.clone(Slimes)) do
			if not rec.dead then rec.dmgBy = {} killSlime(rec, nil) n += 1 end
		end
		notify(admin, n .. " slimes removidos (sem recompensa).", "ok")
	elseif cmd == "slimes_respawn" then
		for _, sp in ipairs(Spawns) do
			if sp.alive then sp.alive.dmgBy = {} killSlime(sp.alive, nil) end
			sp.respawnAt = 0
		end
		notify(admin, "Todos os slimes serão recriados.", "ok")
	elseif cmd == "slimes_toggle" then
		respawnEnabled = not respawnEnabled
		notify(admin, "Respawn de slimes: " .. (respawnEnabled and "LIGADO" or "DESLIGADO"), "ok")
	elseif cmd == "slimes_count" then
		notify(admin, ("Slimes vivos: %d / spawns: %d / respawn: %s"):format(aliveSlimeCount(), #Spawns, respawnEnabled and "ligado" or "desligado"), "ok")
	elseif cmd == "slime_spawn" then
		local kind = args.kind
		if typeof(kind) ~= "string" or not Config.SlimeKinds[kind] then return end
		local _, _, hrp = aliveParts(admin)
		if not hrp then return end
		local lvl = math.clamp(math.floor(num or 1), 1, 100)
		spawnSlime({kind = kind, level = lvl, pos = hrp.Position + hrp.CFrame.LookVector * 12, respawn = 0, temp = true})
		notify(admin, "Slime criado.", "ok")
	end
end)

local function onCharacter(p, char)
	local s = sessions[p]
	if not s then return end
	local waited = 0
	while not s.ready and p.Parent == Players and waited < 30 do task.wait(0.2) waited += 0.2 end
	if not s.ready then return end
	local hum = char:WaitForChild("Humanoid", 10)
	local hrp = char:WaitForChild("HumanoidRootPart", 10)
	if not hum or not hrp then return end
	applyMaxHealth(p, true)
	s.stamina = 100
	s.wantSprint = false
	s.exhaustedUntil = 0
	s.needRecover = false
	hum.WalkSpeed = Config.WALK_SPEED * s.speedMult
	refreshTool(p)
	pushState(p)
end

local function onJoin(p)
	local s = {
		data = defaultData(), ready = false, canSave = false, saving = false,
		stamina = 100, wantSprint = false, exhaustedUntil = 0, needRecover = false, regenAt = 0,
		lastSend = 0, lastSentVal = 100, lastSentFlag = 0, attackReady = 0, skillCd = {}, speedMult = 1,
	}
	sessions[p] = s
	p.CharacterAdded:Connect(function(char) onCharacter(p, char) end)
	local ok, raw = loadData(p)
	if ok then
		s.data = sanitize(raw)
		s.canSave = true
	else
		s.data = defaultData()
		s.canSave = false
	end
	s.ready = true
	pushState(p)
	if not ok then
		notify(p, "Falha ao carregar seus dados: o progresso desta sessão NÃO será salvo.", "err")
	end
	if p.Character then task.spawn(onCharacter, p, p.Character) end
end

Players.PlayerAdded:Connect(onJoin)
for _, p in ipairs(Players:GetPlayers()) do task.spawn(onJoin, p) end

Players.PlayerRemoving:Connect(function(p)
	saveData(p)
	sessions[p] = nil
end)

		task.wait(90)
		for p in pairs(sessions) do task.spawn(saveData, p) end
	end
end)

	if parent then o.Parent = parent end

end

local function openPanel(name)
	if openName == name then
		closePanel(name)
		openName = nil
		return
	end
	if openName then closePanel(openName) end
	openName = name
	local p = panels[name]
	p.token += 1
	p.refresh()
	p.frame.Visible = true
	p.frame.Position = UDim2.new(0.5, 0, 0.5, 16)
	tw(p.frame, {GroupTransparency = 0, Position = UDim2.fromScale(0.5, 0.5)}, 0.25, Enum.EasingStyle.Back)
end

local function owned(id) return table.find(state.inv, id) ~= nil end

local function makeBrowser(panel, mode)
	local content = panel.content
	local list = mk("ScrollingFrame", {Position = UDim2.fromOffset(8, 8), Size = UDim2.new(0, 250, 1, -16), BackgroundTransparency = 1, ScrollBarThickness = 4,
		AutomaticCanvasSize = Enum.AutomaticSize.Y, CanvasSize = UDim2.new()}, content)
	mk("UIListLayout", {Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder}, list)
	local detail = mk("Frame", {Position = UDim2.fromOffset(266, 8), Size = UDim2.new(1, -274, 1, -16), BackgroundColor3 = T.bg2}, content)
	round(detail, 12) stroke(detail, T.panel2, 1.5)
	local sel = nil

	local function render()
		for _, c in ipairs(list:GetChildren()) do if c:IsA("GuiObject") then c:Destroy() end end
		for _, c in ipairs(detail:GetChildren()) do c:Destroy() end
		local ids = {}
		for _, s in ipairs(Config.Swords) do
			if (mode == "shop" and s.shop) or (mode == "inv" and owned(s.id)) then table.insert(ids, s) end
		end
		table.sort(ids, function(a, b)
			local ra, rb = Config.Rarities[a.rarity].order, Config.Rarities[b.rarity].order
			if ra ~= rb then return ra < rb end
			return a.order < b.order
		end)
		local found = false
		for _, s in ipairs(ids) do if s.id == sel then found = true end end
		if not found then sel = ids[1] and ids[1].id or nil end
		for i, s in ipairs(ids) do
			local rar = Config.Rarities[s.rarity]
			local card = mk("TextButton", {Size = UDim2.new(1, -6, 0, 54), BackgroundColor3 = (s.id == sel) and T.panel2 or T.bg2, Text = "", AutoButtonColor = false, LayoutOrder = i}, list)
			round(card, 10) stroke(card, rar.color, (s.id == sel) and 2 or 1, (s.id == sel) and 0 or 0.6)
			mk("Frame", {Size = UDim2.new(0, 5, 1, -12), Position = UDim2.fromOffset(6, 6), BackgroundColor3 = rar.color, BorderSizePixel = 0}, card)
			label(card, s.name, 14, rar.color, Enum.Font.GothamBold, {Position = UDim2.fromOffset(20, 6), Size = UDim2.new(1, -28, 0, 18), TextWrapped = false})
			local sub
			if s.id == state.equipped then sub = "EQUIPADA"
			elseif owned(s.id) then sub = "POSSUÍDA"
			elseif s.shop then sub = rar.label .. "  •  $ " .. fmt(s.price)
			else sub = rar.label end
			label(card, sub, 12, (s.id == state.equipped) and T.good or T.sub, Enum.Font.Gotham, {Position = UDim2.fromOffset(20, 28), Size = UDim2.new(1, -28, 0, 16), TextWrapped = false})
			card.Activated:Connect(function() sel = s.id render() end)
		end
		local s = sel and Config.SwordsById[sel]
		if not s then
			label(detail, mode == "inv" and "Nenhuma espada." or "Nada à venda.", 16, T.sub, Enum.Font.GothamBold, {Size = UDim2.fromScale(1, 1), TextXAlignment = Enum.TextXAlignment.Center})
			return
		end
		local rar = Config.Rarities[s.rarity]
		label(detail, s.name, 22, rar.color, Enum.Font.GothamBlack, {Position = UDim2.fromOffset(14, 10), Size = UDim2.new(1, -28, 0, 28), TextWrapped = false})
		label(detail, rar.label:upper(), 12, rar.color, Enum.Font.GothamBold, {Position = UDim2.fromOffset(14, 38), Size = UDim2.new(1, -28, 0, 16)})
		local rows = {
			{"Dano", s.damage, s.damage / 340, tostring(s.damage)},
			{"Recarga", s.cooldown, (1 / s.cooldown) / 2.6, ("%.2fs"):format(s.cooldown)},
			{"Alcance", s.range, s.range / 13, tostring(s.range)},
			{"Crítico", s.crit, s.crit / 0.2, math.floor(s.crit * 100 + 0.5) .. "%"},
		}
		for i, r in ipairs(rows) do
			local y = 62 + (i - 1) * 26
			label(detail, r[1], 13, T.sub, Enum.Font.GothamBold, {Position = UDim2.fromOffset(14, y), Size = UDim2.fromOffset(70, 20)})
			local bg = mk("Frame", {Position = UDim2.fromOffset(88, y + 5), Size = UDim2.new(1, -160, 0, 10), BackgroundColor3 = T.bg}, detail)
			round(bg, 5)
			local fl = mk("Frame", {Size = UDim2.fromScale(math.clamp(r[3], 0.04, 1), 1), BackgroundColor3 = rar.color}, bg)
			round(fl, 5)
			label(detail, r[4], 13, T.text, Enum.Font.GothamBold, {AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -14, 0, y), Size = UDim2.fromOffset(56, 20), TextXAlignment = Enum.TextXAlignment.Right})
		end
		local sk = s.skill
		local box = mk("Frame", {Position = UDim2.fromOffset(14, 172), Size = UDim2.new(1, -28, 0, 112), BackgroundColor3 = T.panel}, detail)
		round(box, 10) stroke(box, sk.color, 1.5, 0.3)
		label(box, "HABILIDADE ESPECIAL", 11, sk.color, Enum.Font.GothamBlack, {Position = UDim2.fromOffset(12, 8), Size = UDim2.new(1, -24, 0, 14)})
		label(box, sk.name .. "   •   recarga " .. sk.cd .. "s", 15, T.text, Enum.Font.GothamBold, {Position = UDim2.fromOffset(12, 26), Size = UDim2.new(1, -24, 0, 20), TextWrapped = false})
		label(box, sk.desc, 13, T.sub, Enum.Font.Gotham, {Position = UDim2.fromOffset(12, 50), Size = UDim2.new(1, -24, 0, 56), TextYAlignment = Enum.TextYAlignment.Top})
		local btn = mk("TextButton", {AnchorPoint = Vector2.new(0.5, 1), Position = UDim2.new(0.5, 0, 1, -12), Size = UDim2.new(1, -28, 0, 44), Font = Enum.Font.GothamBlack,
			TextSize = 16, TextColor3 = Color3.new(1, 1, 1), BackgroundColor3 = T.accent, AutoButtonColor = true}, detail)
		round(btn, 10)
		if s.id == state.equipped then
			btn.Text, btn.BackgroundColor3, btn.AutoButtonColor = "EQUIPADA", T.panel2, false
		elseif owned(s.id) then
			btn.Text, btn.BackgroundColor3 = "EQUIPAR", T.good
			btn.Activated:Connect(function() rEquip:FireServer(s.id) end)
		elseif s.shop then
			if state.coins >= s.price then
				btn.Text, btn.BackgroundColor3 = "COMPRAR  —  $ " .. fmt(s.price), T.accent
				btn.Activated:Connect(function() rBuy:FireServer(s.id) end)
			else
				btn.Text, btn.BackgroundColor3, btn.AutoButtonColor = "MOEDAS INSUFICIENTES  ($ " .. fmt(s.price) .. ")", T.bad, false
				btn.TextSize = 14
			end
		else
			btn.Text, btn.BackgroundColor3, btn.AutoButtonColor = "NÃO DISPONÍVEL", T.panel2, false
		end
	end
	panel.refresh = render
	return render
end

local dlgToken = 0
local dlgNpc, dlgPos = nil, nil

local function closeDialogue()
	dlgNpc, dlgPos = nil, nil
	dlgToken += 1
	local tk = dlgToken
	tw(dlg, {GroupTransparency = 1}, 0.2)
	task.delay(0.22, function() if dlgToken == tk then dlg.Visible = false end end)
end

rDialogue.OnClientEvent:Connect(function(payload)
	if typeof(payload) ~= "table" then return end
	dlgToken += 1
	local tk = dlgToken
	dlgNpc, dlgPos = payload.npc, payload.pos
	dlg.Visible = true
	tw(dlg, {GroupTransparency = 0}, 0.2)
	dlgName.Text = payload.npc
	dlgText.Text = payload.text
	dlgText.MaxVisibleGraphemes = 0
	for _, c in ipairs(dlgOptions:GetChildren()) do if c:IsA("GuiObject") then c:Destroy() end end
	for i, o in ipairs(payload.options) do
		local b = mk("TextButton", {Size = UDim2.new(1, -6, 0, 32), BackgroundColor3 = T.panel2, Text = "  " .. o.label, Font = Enum.Font.GothamBold, TextSize = 14,
			TextColor3 = T.text, TextXAlignment = Enum.TextXAlignment.Left, AutoButtonColor = true, LayoutOrder = i}, dlgOptions)
		round(b, 8) stroke(b, T.accent, 1, 0.6)
		b.Activated:Connect(function()
			rChoice:FireServer(payload.npc, o.id)
			if o.close then closeDialogue() end
		end)
	end
	task.spawn(function()
		local n = #dlgText.Text
		for i = 1, n do
			if dlgToken ~= tk then return end
			dlgText.MaxVisibleGraphemes = i
			task.wait(0.012)
		end
		dlgText.MaxVisibleGraphemes = -1
	end)
end)

task.spawn(function()
	while true do
		task.wait(0.5)
		if dlgNpc and dlgPos then
			local char = player.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if not hrp or (hrp.Position - dlgPos).Magnitude > 28 then closeDialogue() end
		end
	end
end)

local function fxPart(size, cf, color, shape, trans)
	local p = mk("Part", {Anchored = true, CanCollide = false, CanQuery = false, CanTouch = false, CastShadow = false, Material = Enum.Material.Neon,

local FX = {}
FX.ring = function(e)
	local p = fxPart(Vector3.new(0.5, 2, 2), CFrame.new(e.p) * CFrame.Angles(0, 0, math.pi / 2), e.c, Enum.PartType.Cylinder, 0.35)
	TweenService:Create(p, TweenInfo.new(0.5), {Size = Vector3.new(0.5, e.r * 2, e.r * 2), Transparency = 1}):Play()
	Debris:AddItem(p, 0.6)
end
FX.slash = function(e)
	local p = fxPart(Vector3.new(7 * (e.s or 1), 0.25, 0.9), e.cf * CFrame.Angles(0, 0, math.rad(70)), e.c, nil, 0.2)
	TweenService:Create(p, TweenInfo.new(0.22), {CFrame = e.cf * CFrame.Angles(0, 0, math.rad(-70)), Transparency = 1}):Play()
	Debris:AddItem(p, 0.3)
end
FX.beam = function(e)
	local p = fxPart(Vector3.new(e.h, e.r * 2, e.r * 2), CFrame.new(e.p + Vector3.new(0, e.h / 2, 0)) * CFrame.Angles(0, 0, math.pi / 2), e.c, Enum.PartType.Cylinder, 0.3)
	TweenService:Create(p, TweenInfo.new(0.6), {Transparency = 1, Size = Vector3.new(e.h, e.r * 0.4, e.r * 0.4)}):Play()
	Debris:AddItem(p, 0.7)
end
FX.bolt = function(e)
	local a, b = e.a, e.b
	local prev = a
	for i = 1, 6 do
		local pt = a:Lerp(b, i / 6)
		if i < 6 then pt += Vector3.new(math.random(-10, 10) / 10, math.random(-10, 10) / 10, math.random(-10, 10) / 10) * 1.5 end
		local len = (pt - prev).Magnitude
		if len > 0.05 then
			local p = fxPart(Vector3.new(0.4, 0.4, len), CFrame.lookAt((prev + pt) / 2, pt), e.c, nil, 0)
			TweenService:Create(p, TweenInfo.new(0.3), {Transparency = 1}):Play()
			Debris:AddItem(p, 0.35)
		end
		prev = pt
	end
end
FX.burst = function(e)
	for _ = 1, math.min(e.n or 12, 30) do
		local p = fxPart(Vector3.new(0.6, 0.6, 0.6), CFrame.new(e.p), e.c, Enum.PartType.Ball, 0.1)
		local dir = Vector3.new(math.random(-10, 10), math.random(2, 12), math.random(-10, 10)).Unit * math.random(4, 10)
		TweenService:Create(p, TweenInfo.new(0.5, Enum.EasingStyle.Quad), {Position = e.p + dir, Transparency = 1, Size = Vector3.new(0.1, 0.1, 0.1)}):Play()
		Debris:AddItem(p, 0.6)
	end
end
FX.dash = function(e)
	local len = (e.b - e.a).Magnitude
	if len < 0.5 then return end
	local p = fxPart(Vector3.new(2.5, 2.5, len), CFrame.lookAt((e.a + e.b) / 2, e.b), e.c, nil, 0.4)
	TweenService:Create(p, TweenInfo.new(0.35), {Transparency = 1, Size = Vector3.new(0.2, 0.2, len)}):Play()
	Debris:AddItem(p, 0.4)
end
FX.wave = function(e)
	local len = (e.b - e.a).Magnitude
	local p = fxPart(Vector3.new(e.w, 0.6, len), CFrame.lookAt((e.a + e.b) / 2, e.b), e.c, nil, 0.3)
	TweenService:Create(p, TweenInfo.new(0.5), {Transparency = 1}):Play()
	Debris:AddItem(p, 0.6)
end
